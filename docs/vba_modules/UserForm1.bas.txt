Private Sub cbDogumTarihiHesapla_Click()

    ' Kodun daha okunaklı ve hızlı çalışması için "Hesaplama" sayfası ile çalışacağımızı belirtiyoruz.
    With ThisWorkbook.Worksheets("Hesaplama")
    
        ' TextBox değerlerini ilgili hücrelere ata.
        ' .Value kullanarak hücrenin değerini değiştiriyoruz.
        ' Val() fonksiyonu, TextBox'a girilen metni sayısal bir değere çevirir.
        ' Bu, olası hataları önlemek için iyi bir pratiktir.
        .Range("AD3").Value = Val(Me.txtGun1.Value)
        .Range("AE3").Value = Val(Me.txtGun2.Value)
        .Range("AF3").Value = Val(Me.txtAy1.Value)
        .Range("AG3").Value = Val(Me.txtAy2.Value)
        .Range("AH3").Value = Val(Me.txtYil1.Value)
        .Range("AI3").Value = Val(Me.txtYil2.Value)
        .Range("AJ3").Value = Val(Me.txtYil3.Value)
        .Range("AK3").Value = Val(Me.txtYil4.Value)
        
    End With

    ' Değerleri hücrelere yazma işlemi bittikten sonra "HesaplamaYap" makrosunu çalıştır.
    ' Call komutunu kullanabilir veya doğrudan makro adını yazabilirsiniz.
    Module7.HesaplamaYap
 
   txtDogumTarihiSonuc.Value = Worksheets("Hesaplama").Range("AF11").Value
