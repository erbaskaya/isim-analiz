Sub ArapcaIsmiTemizleVeAltAltaYazCocuk1()


    Dim metin As String
    Dim i As Integer
    
    ' Önce O7:O20 aralığını temizle
    Range("O7:O20").ClearContents
    
    ' C6 hücresindeki metni al
    metin = Range("O6").Value
    
    ' Metin boş değilse her harfi alt alta yaz
    If Len(metin) > 0 Then
        For i = 1 To Len(metin)
            Range("O" & 6 + i).Value = Mid(metin, i, 1)
        Next i
    End If

End Sub
